package synthrepo // github.com/GiulioSavini/gruntled/internal/testsupport/synthrepo

type ErrorKind int
const BadOutputRef ErrorKind = iota + 1 // a dependency.X.outputs.Y naming an output the target module does not declare
func (k ErrorKind) String() string       // "bad-output-ref"

type Spec struct {
	Units            int         // >= 1
	IncludeDepth     int         // >= 1: directory levels between repo root (root.hcl) and each unit dir
	DependencyFanout int         // >= 0: unit i gets min(DependencyFanout, i) dependencies on lower-index units (acyclic by construction)
	Seed             int64
	Errors           []ErrorKind // each entry injects one distinct instance
}
func (s Spec) Validate() error

type File struct {
	Path    string // slash-separated, repo-relative
	Content []byte
}
type Tree []File              // sorted by Path
func (t Tree) Digest() string // hex SHA-256 over, for each file in order: Path + "\x00" + Content + "\x00"

type ExpectedDiagnostic struct {
	Code       diagnostic.Code    // diagnostic.CodeUnknownOutput
	Pos        repograph.Position // <unit>/terragrunt.hcl, 1-based line and byte column of the "d" in dependency.<dep>.outputs.<out>
	Unit       repograph.RepoPath // unit containing the reference
	Dependency string             // dependency block label, e.g. "dep_003"
	Target     repograph.RepoPath // unit the dependency points at
	Output     string             // the undeclared output name
}
type Manifest struct {
	Units    []repograph.RepoPath // sorted
	Expected []ExpectedDiagnostic // sorted by Pos
}

func Render(spec Spec) (Tree, Manifest, error)             // pure: no I/O
func Generate(spec Spec, destDir string) (Manifest, error) // destDir must exist, be a directory and be empty; never deletes anything
</interfaces>

    Render algorithm. Consume the RNG in exactly this order and document it in a comment; changing the order changes every digest.
    1. `width := max(3, len(strconv.Itoa(Units-1)))`.
    2. For i in 0..Units-1, in order:
       - name `fmt.Sprintf("unit-%0*d", width, i)`
       - IncludeDepth-1 group segments, each `fmt.Sprintf("g%d", rng.IntN(2))`
       - unit path = `path.Join(groups..., name)`
       - output count `1 + rng.IntN(3)`, with outputs named `out_0`…`out_{n-1}`
    3. For i in 0..Units-1, in order:
       - `k := min(DependencyFanout, i)`
       - targets = `rng.Perm(i)[:k]` sorted ascending
       - for each target j, in order: dependency name `fmt.Sprintf("dep_%0*d", width, j)`; referenced output `out_<rng.IntN(len(outputs[j]))>`; input key `in_<depname>_<output>`
       This yields one reference per dependency.
    4. Injections: keep the list of all references in (unit index, dependency order). For each BadOutputRef, in order, pick `rng.IntN(len(remaining))` from the not-yet-corrupted ones and set its output to `fmt.Sprintf("missing_out_%d", ordinal)`. Keep its input key and consumer variable name unchanged, so the mutation is a single token, like a real rename bug.

    File contents use "\n" line endings, 2-space indentation and a trailing newline:
    - `root.hcl`:
      ```
      # Generated by gruntled synthrepo. Do not edit.
      inputs = {
        synthrepo_seed = "<seed>"
      }
      ```
    - `<unit>/terragrunt.hcl`: the header comment; then `include "root" {` / `  path = find_in_parent_folders("root.hcl")` / `}`; then, for each dependency, a blank line and `dependency "<dep>" {` / `  config_path = "<relPath(unit, target)>"` / `}`; then, only if there are dependencies, a blank line and `inputs = {` / `  <key> = dependency.<dep>.outputs.<out>` per reference / `}`.
    - `<unit>/main.tf`: the header comment; for each input key, in order, a blank line and `variable "<key>" {` / `  type = string` / `}`; for each output, a blank line and `output "<out>" {` / `  value = "<unit path>/<out>"` / `}`.

    Build the content with a `bytes.Buffer`. Record the byte offset of each `dependency.` traversal in the inputs block while writing it, then convert it with lineCol into a repograph.Position whose file is `<unit>/terragrunt.hcl`.

    relPath and lineCol are pure string helpers. Use no filepath here; render.go does no I/O at all. Sort the Tree by Path, sort Manifest.Units with RepoPath.Compare, and sort Expected by Pos.Compare.
