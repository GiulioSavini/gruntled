package repograph  // github.com/GiulioSavini/gruntled/internal/domain/repograph

// RepoPath: a repository-relative, slash-separated, cleaned path. Struct with an unexported field, so an unvalidated value cannot be produced by conversion. Comparable, so it is usable as a map key.
type RepoPath struct{ /* p string */ }
func NewRepoPath(s string) (RepoPath, error) // rules: non-empty; path.Clean(s)==s; not absolute ("/" prefix); no "\\"; not ".." and no "../" prefix; "." (repo root) allowed
func MustRepoPath(s string) RepoPath        // panics on invalid input; for tests and generator constants
func (p RepoPath) String() string
func (p RepoPath) IsZero() bool
func (p RepoPath) Compare(q RepoPath) int   // strings.Compare

// Position: 1-based line and column (column counted in bytes) inside a repo-relative file.
type Position struct{ /* file RepoPath; line, column int */ }
func NewPosition(file RepoPath, line, column int) (Position, error) // file non-zero, line>=1, column>=1
func (p Position) File() RepoPath
func (p Position) Line() int
func (p Position) Column() int
func (p Position) Compare(q Position) int // file, then line, then column
func (p Position) String() string         // "file:line:column"

// Surface: the public variable and output names of a Module (design doc §5.3).
type Surface struct{ /* variables, outputs []string, both sorted */ }
func NewSurface(variables, outputs []string) (Surface, error) // rejects an empty name or a duplicate within a list; stores sorted copies
func (s Surface) Variables() []string // sorted copy
func (s Surface) Outputs() []string   // sorted copy
func (s Surface) HasVariable(name string) bool // slices.BinarySearch
func (s Surface) HasOutput(name string) bool

type Module struct{ /* path RepoPath; surface Surface */ }
func NewModule(path RepoPath, surface Surface) (Module, error) // path non-zero
func (m Module) Path() RepoPath
func (m Module) Surface() Surface

// Dependency: a `dependency "<name>" { config_path = ... }` block, with target already resolved to a unit path.
type Dependency struct{ /* name string; target RepoPath; pos Position */ }
func NewDependency(name string, target RepoPath, pos Position) (Dependency, error)
func (d Dependency) Name() string
func (d Dependency) Target() RepoPath
func (d Dependency) Pos() Position

// Reference: one `dependency.<Dependency>.outputs.<Output>` traversal; Pos is the first byte of `dependency`.
type Reference struct{ /* dependency, output string; pos Position */ }
func NewReference(dependency, output string, pos Position) (Reference, error)
func (r Reference) Dependency() string
func (r Reference) Output() string
func (r Reference) Pos() Position

type UnitStatus int
const (
	StatusResolved UnitStatus = iota + 1 // module known and local
	StatusUnknown                        // could not be resolved offline; analyzers must stay silent (design doc §7)
)
func (s UnitStatus) String() string // "resolved" | "unknown"

type Unit struct{ /* unexported */ }
func NewResolvedUnit(path, module RepoPath, deps []Dependency, refs []Reference) (Unit, error)
	// path and module non-zero; dependency names unique within the unit; deps sorted by Name; refs sorted by Pos, then Dependency, then Output; both cloned
func NewUnknownUnit(path RepoPath, reason string) (Unit, error) // reason non-empty; no deps and no refs
func (u Unit) Path() RepoPath
func (u Unit) Status() UnitStatus
func (u Unit) Module() (RepoPath, bool)   // false when Unknown
func (u Unit) UnknownReason() string
func (u Unit) Dependencies() []Dependency // sorted copy
func (u Unit) Dependency(name string) (Dependency, bool)
func (u Unit) References() []Reference    // sorted copy

type UnitReference struct {
	Unit      RepoPath
	Reference Reference
}

type RepositoryGraph struct{ /* units []Unit sorted; modules []Module sorted; lookup maps (never ranged) */ }
func NewRepositoryGraph(units []Unit, modules []Module) (*RepositoryGraph, error)
	// sorts both; errors: *DuplicateUnitError{Path}, *DuplicateModuleError{Path},
	// *MissingModuleError{Unit, Module} when a resolved unit's module is absent from modules.
	// A dependency whose Target is not a unit in the graph is NOT an error (a future GRT002 diagnoses it).
func (g *RepositoryGraph) Units() []Unit
func (g *RepositoryGraph) Unit(p RepoPath) (Unit, bool)
func (g *RepositoryGraph) Modules() []Module
func (g *RepositoryGraph) Module(p RepoPath) (Module, bool)
func (g *RepositoryGraph) ModuleOf(unit RepoPath) (Module, bool)                    // false if the unit is absent or Unknown
func (g *RepositoryGraph) DependencyTarget(unit RepoPath, dep string) (Unit, bool) // hop 1 of GRAPH-04; false if the unit, the dependency or the target unit is missing
func (g *RepositoryGraph) References() []UnitReference                            // every reference of every unit, sorted by Reference.Pos, then Unit, then Dependency, then Output

package diagnostic  // github.com/GiulioSavini/gruntled/internal/domain/diagnostic  (imports repograph)

type Code string
const (
	CodeUnknownOutput Code = "GRT001" // dependency.X.outputs.Y names an output the target module does not declare
	CodeSyntaxError   Code = "GRT100" // invalid HCL
)
func ParseCode(s string) (Code, error) // valid iff "GRT" followed by exactly 3 ASCII digits; hand-written check, no regexp
func (c Code) Valid() bool

type Severity int
const (
	SeverityError Severity = iota + 1
	SeverityWarning
)
func (s Severity) String() string // "error" | "warning"

type Diagnostic struct{ /* code Code; severity Severity; pos repograph.Position; message string */ }
func New(code Code, severity Severity, pos repograph.Position, message string) (Diagnostic, error)
	// code.Valid(); severity is one of the constants; pos non-zero (File non-zero); message non-empty
func (d Diagnostic) Code() Code
func (d Diagnostic) Severity() Severity
func (d Diagnostic) Pos() repograph.Position
func (d Diagnostic) Message() string
func (d Diagnostic) Key() Key
func Compare(a, b Diagnostic) int // Pos (file, line, column), then Code, then Severity, then Message

type Key struct { // comparable; stable identity used later for Diff and at-most-once notifications
	Code    Code
	File    string
	Line    int
	Column  int
	Message string
}

type Set struct{ /* items []Diagnostic, canonical order, unique by Key */ }
func NewSet(ds ...Diagnostic) Set // clone, sort with Compare, drop later duplicates by Key
func (s Set) All() []Diagnostic   // copy in canonical order
func (s Set) Len() int
func (s Set) HasErrors() bool
func (s Set) Equal(o Set) bool
func Diff(prev, next Set) (added, removed Set) // added = in next not prev; removed = in prev not next (by Key)
