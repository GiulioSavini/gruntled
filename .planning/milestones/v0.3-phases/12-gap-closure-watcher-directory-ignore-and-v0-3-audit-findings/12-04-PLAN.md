```go
package presenter

// escapeTerm returns s with every byte sequence that a terminal could interpret
// replaced by a visible escape; s is returned unchanged (no allocation) when
// nothing needs escaping.
//   invalid UTF-8 byte            -> \xNN
//   U+0000-U+001F, U+007F         -> \xNN
//   U+0080-U+009F (C1)            -> \uNNNN
//   unicode.Cf, U+2028, U+2029    -> \uNNNN  (\UNNNNNNNN above U+FFFF)
// hex is lowercase; everything else, including '\\', is copied verbatim.
func escapeTerm(s string) string

// escapeJSON rewrites, in the output of encoding/json, every rune that a
// terminal could act on but encoding/json leaves raw: U+007F, U+0080-U+009F,
// unicode.Cf, U+2028, U+2029 => \uXXXX (lowercase hex; runes above U+FFFF as a
// \uXXXX\uXXXX surrogate pair). These runes can only occur inside JSON
// strings (all structural bytes are ASCII), so the result decodes to the same
// values. Returns b itself (no allocation) when every byte is < 0x7f.
func escapeJSON(b []byte) []byte
```
