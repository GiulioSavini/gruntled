package tfsurface

// maxTypeDepth bounds the canonical renderer; deeper types are unknown.
const maxTypeDepth = 100

// typeSig returns ty's canonical string, or ("", false) for a capsule or
// unrecognised kind or a type nested deeper than maxTypeDepth. It never panics.
func typeSig(ty cty.Type) (string, bool)

// maxJSONTypeBytes caps a decoded .tf.json type string; longer is unknown.
const maxJSONTypeBytes = 64 << 10

// maxJSONTypeNesting bounds the ( [ { nesting of a .tf.json type string
// before it reaches typeexpr (which parses it with the recursive native
// parser, bypassing CheckJSONDepth).
const maxJSONTypeNesting = 2*maxTypeDepth + 2

// typeAttrSig parses a type constraint expression with
// typeexpr.TypeConstraintWithDefaults (defaults discarded) and renders it;
// ("", false) on any error diagnostic. For a .tf.json file (fromJSON) it
// first reads the raw string from src[expr.Range()] with encoding/json
// (a non-string JSON value is unknown) and returns ("", false), without
// calling typeexpr, when the decoded string exceeds maxJSONTypeBytes,
// jsonTypeNesting exceeds maxJSONTypeNesting, or
// hclconv.CheckNativeExprDepth refuses it.
func typeAttrSig(expr hcl.Expression, src []byte, fromJSON bool) (string, bool)

// jsonTypeNesting is a linear byte scan returning the maximum ( [ {
// nesting depth of s (brackets inside quoted defaults count too, which only
// makes the answer more conservative).
func jsonTypeNesting(s string) int

// parseTypeFact is the seam facts.go calls; production value wraps
// typeAttrSig with a narrow recover (panic -> ("", false)). The recover
// cannot catch a stack overflow, which Go turns into a fatal error: deep
// input is stopped only by the pre-scans (CheckNativeDepth on the file and,
// for .tf.json, the 64 KiB cap plus CheckNativeExprDepth on the decoded type
// string).
var parseTypeFact = safeTypeAttrSig

// isOverrideFile reports whether a kept module file name is a Terraform
// override file: base name "override" or ending in "_override" after
// stripping .tf, .tf.json, .tofu or .tofu.json.
func isOverrideFile(name string) bool
