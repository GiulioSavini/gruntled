```
#!/usr/bin/env bash
# release-notes.sh <tag> [changelog]
# Prints the CHANGELOG section for tag vX.Y.Z[-suffix] (heading "## [X.Y.Z] - date")
# to stdout. Exit 1 when the section is absent or empty, 2 on a malformed tag or usage.
# release.yml feeds the result to gh release create --notes-file; ci.yml dry-runs it.
set -euo pipefail
[ "$#" -ge 1 ] && [ "$#" -le 2 ] || { echo "usage: $0 <tag> [changelog]" >&2; exit 2; }
tag=$1
file=${2:-CHANGELOG.md}
if ! [[ "$tag" =~ ^v[0-9]+\.[0-9]+\.[0-9]+(-[0-9A-Za-z.-]+)?$ ]]; then
  echo "release-notes: tag '$tag' does not match vX.Y.Z[-suffix]" >&2; exit 2
fi
[ -r "$file" ] || { echo "release-notes: cannot read $file" >&2; exit 1; }
want="## [${tag#v}] - "
body=$(
  found=0
  while IFS= read -r line || [ -n "$line" ]; do
    if [ "$found" -eq 1 ]; then
      case "$line" in "## ["*|"["*"]: http"*) break ;; esac
      printf '%s\n' "$line"
    elif [ "${line:0:${#want}}" = "$want" ]; then
      found=1
    fi
  done < "$file"
)
# trim trailing blank lines (fixed awk program; the tag is not part of it)
body=$(printf '%s\n' "$body" | awk '{l[NR]=$0} END{n=NR; while (n>0 && l[n] ~ /^[[:space:]]*$/) n--; for (i=1;i<=n;i++) print l[i]}')
if [ -z "${body//[[:space:]]/}" ]; then
  echo "release-notes: no section [${tag#v}] in $file" >&2; exit 1
fi
printf '%s\n' "$body"
```
The heading compare is a fixed-length substring equality (`${line:0:N}` = `$want`), so no character
of the tag is interpreted by `case`, `[[ ]]`, grep or awk. The tag regex is the same one as the
release guard. Add `scripts/release-notes.sh` to the shellcheck run if one exists in ci.yml (none
today: skip). Commit `feat(18-03): add scripts/release-notes.sh`.
