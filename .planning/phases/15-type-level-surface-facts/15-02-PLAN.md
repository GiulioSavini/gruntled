```go
package tfsurface

// maxTypeDepth bounds the canonical renderer; deeper types are unknown.
const maxTypeDepth = 100

// typeSig returns ty's canonical string, or ("", false) for a capsule or
// unrecognised kind or a type nested deeper than maxTypeDepth. It never panics.
func typeSig(ty cty.Type) (string, bool)

// typeAttrSig parses a type constraint expression with
// typeexpr.TypeConstraintWithDefaults (defaults discarded) and renders it;
// ("", false) on any error diagnostic.
func typeAttrSig(expr hcl.Expression) (string, bool)

// isOverrideFile reports whether a kept module file name is a Terraform
// override file: base name "override" or ending in "_override" after
// stripping .tf, .tf.json, .tofu or .tofu.json.
func isOverrideFile(name string) bool
```
