```go
package ipc

const (
    ProtocolVersion = 1
    LockName = "lock"; SockName = "sock"; DumpName = "report"   // files inside statusfile.Dir(root, env)
    OpPing = "ping"; OpReport = "report"
    StateIndexing = "indexing"; StateReady = "ready"; StateFailed = "failed"
    MaxSockPath = 103 // darwin sun_path 104 incl. NUL
)
var (
    ErrHeld = errors.New("ipc: lock held")          // another daemon owns the lock
    ErrNoDaemon = errors.New("ipc: no daemon")      // ENOENT / ECONNREFUSED on dial
    ErrUnsupported = errors.New("ipc: sockets unsupported on this platform")
)
type VersionError struct{ Daemon, Client int }       // protocol mismatch; Error() names both

// Info is static per daemon process.
type Info struct { PID int; Version, Root, StatusPath string }
// Snapshot is the pre-rendered result of one successful index. []byte fields are base64 in JSON => byte exact.
type Snapshot struct {
    State     string // StateReady | StateFailed
    LastError string // sanitised, set when StateFailed after a good index
    HasErrors bool
    Generation uint64
    Text, Summary, JSON, SARIF []byte
}

func CheckSockPath(path string) error // nil if len(path) <= MaxSockPath, else error naming length and XDG_RUNTIME_DIR/TMPDIR

type Lock struct{ /* *os.File */ }
func TryLock(path string) (*Lock, error)  // single attempt; ErrHeld when held; creates file 0600; flock (linux||darwin), CreateFile share 0 (windows), no-op on other GOOS
func LockRetry(path string, attempts int, sleep func(time.Duration)) (*Lock, error) // attempts>=1; sleeps 20ms between ErrHeld tries; sleep injected
func (*Lock) Release() error              // closes handle; does NOT delete the file
func Probe(path string) (held bool, err error) // true iff another holder exists; if the file is absent => false,nil; acquires+releases when free

// linux || darwin only; other GOOS: stubs returning ErrUnsupported
type Listener struct{ /* *os.File, path */ }
func Listen(path string) (*Listener, error) // caller MUST hold the Lock. Lstat: ModeSocket => remove (stale); other type => error, never delete; bind; chmod 0600; listen
type Server struct{ /* ... */ }
func Serve(ln *Listener, info Info, get func() *Snapshot) *Server // get() nil => state indexing; handlers only call get()
func (*Server) Close() error // idempotent: close listener (wakes Accept), wait handlers, remove socket path
func Query(sock, op string, timeout time.Duration) (Info, *Snapshot, error) // *Snapshot nil iff daemon is indexing; ErrNoDaemon; *VersionError

// all platforms (windows report transport; tested on linux)
func WriteDump(w *statusfile.Writer, info Info, s *Snapshot) error // JSON, atomic via Writer; s may be nil => indexing
func ReadDump(path string) (Info, *Snapshot, error)
```
Wire: one request line `{"v":1,"op":"ping"|"report"}\n` (server reads <= 4 KiB, 2 s deadline, one request per connection); one response line `{"v":1,"ok":true,"info":{...},"state":"indexing|ready|failed","snapshot":{...}}` or `{"v":1,"ok":false,"error":"..."}`. Client: connect timeout + 2 s deadline, response capped at 64 MiB via io.LimitReader, v mismatch => *VersionError.
