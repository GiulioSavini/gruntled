package terragrunt

// TestZZMeasure prints a status/reason histogram per corpus in GRUNTLED_MEASURE
// (colon-separated dirs): units, refs, checkable refs (target module surface
// known), missing (would-be GRT001), diagnostics. NOT committed.
func TestZZMeasure(t *testing.T) {
	for _, dir := range strings.Split(os.Getenv("GRUNTLED_MEASURE"), ":") {
		root, err := os.OpenRoot(dir)
		if err != nil { t.Fatal(err) }
		res := build(t, root.FS())
		hist := map[string]int{}
		for _, u := range res.Graph.Units() { hist["unit "+u.Status().String()+" "+u.UnknownReason()]++ }
		for _, m := range res.Graph.Modules() { if _, ok := m.Surface(); !ok { hist["module-unknown "+m.UnknownReason()]++ } }
		checkable, missing := 0, 0
		for _, ur := range res.Graph.References() {
			tu, ok := res.Graph.DependencyTarget(ur.Unit, ur.Reference.Dependency()); if !ok { continue }
			mod, ok := res.Graph.ModuleOf(tu.Path()); if !ok { continue }
			s, ok := mod.Surface(); if !ok { continue }
			checkable++
			if !s.HasOutput(ur.Reference.Output()) { missing++ }
		}
		fmt.Printf("%s units=%d refs=%d checkable=%d missing=%d diags=%d\n", dir, len(res.Graph.Units()), len(res.Graph.References()), checkable, missing, res.Diagnostics.Len())
		keys := slices.Sorted(maps.Keys(hist))
		for _, k := range keys { fmt.Printf("  %-60s %d\n", k, hist[k]) }
		root.Close()
	}
}

hclsyntax facts this plan relies on (verified in hcl/v2@v2.25.0 hclsyntax/parser.go; re-check them in `$(go env GOMODCACHE)/github.com/hashicorp/hcl/v2@v2.25.0/hclsyntax/`):
- parseTernaryConditional (parser.go:491) parses the condition with parseBinaryOps, and then, after `?`, parses BOTH the true and the false branch with p.ParseExpression(), which re-enters parseTernaryConditional. Every `?` is therefore one recursion level that is released only when the whole conditional expression ends. That happens at the enclosing bracket's close, at a comma, or at a newline when newlines are significant.
- Newlines are significant only in a body (the file top level and block bodies) and in an object constructor `{ ... }` that is NOT a for-expression. parseObjectCons (parser.go:1415) peeks for the `for` keyword and switches to finishParsingForExpr with PushIncludeNewlines(false). Parens, brackets (tuples, index keys), function-call args, `${`/`%{` template sequences and for-expressions all ignore newlines (PushIncludeNewlines(false) at :881, :967, :1238, :1342, :1589, parser_template.go:414/:486).
- peeker.nextToken turns a line comment (`#`/`//`) whose bytes end in "\n" into a newline when newlines are significant, so such a TokenComment counts as a newline.
- parseBinaryOps loops (left-associative) and does not recurse per operator. A long `+` chain therefore does not overflow the parser, but it builds a left-leaning BinaryOpExpr tree as deep as the chain, which later recursive walks (refs extraction, Variables) descend.
