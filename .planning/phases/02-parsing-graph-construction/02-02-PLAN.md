TARGET ports API (Plans 03-05 implement it; use exactly these names):
```go
package ports

// UnitLoader discovers every unit in the repository and returns its structural configuration.
// Per-file problems (syntax errors, dynamic paths, remote sources) are NOT errors: they are
// reported through UnknownReason fields and Diagnostics. error is reserved for context
// cancellation and failures that make the whole load meaningless (repository root unreadable).
type UnitLoader interface {
    LoadUnits(ctx context.Context) (LoadResult, error)
}

type LoadResult struct {
    Units       []UnitConfig            // sorted by Path
    Diagnostics []diagnostic.Diagnostic // GRT100 for terragrunt.hcl / include files, file-level (zero Unit)
}

// UnitConfig is one unit. Exactly one state applies, checked in this order:
//   ConfigUnknownReason != ""  -> config-unknown: every other field is ignored.
//   ModuleUnknownReason != ""  -> module-unknown: Module is ignored; Dependencies and References are kept.
//   otherwise                  -> resolved: Module is the local module directory (== Path when terraform.source is absent).
type UnitConfig struct {
    Path                repograph.RepoPath
    ConfigUnknownReason string
    ModuleUnknownReason string
    Module              repograph.RepoPath
    Dependencies        []repograph.Dependency // include-merged, unique names; may contain unresolved dependencies
    References          []repograph.Reference  // from the unit's entire effective body, including merged include files
}

// SurfaceReader reads a local module directory's variable and output names.
type SurfaceReader interface {
    ReadSurface(ctx context.Context, module repograph.RepoPath) (SurfaceResult, error)
}

type SurfaceResult struct {
    Surface       repograph.Surface       // valid only when UnknownReason == ""
    UnknownReason string                  // non-empty => surface unknown (dir missing, no .tf files, syntax error, ...)
    Diagnostics   []diagnostic.Diagnostic // GRT100 for broken module files, file-level
}
```
TARGET indexing API:
```go
package indexing

type Result struct {
    Graph       *repograph.RepositoryGraph
    Diagnostics diagnostic.Set
}

// Build loads every unit, reads each distinct module surface once, and assembles the graph.
func Build(ctx context.Context, units ports.UnitLoader, surfaces ports.SurfaceReader) (Result, error)

// Error wraps a failure with the Build stage it happened in ("load-units", "read-surface",
// "assemble"); Unwrap returns the cause so errors.Is/As work.
type Error struct { Stage string; Path string; Err error }
func (e *Error) Error() string
func (e *Error) Unwrap() error
```
