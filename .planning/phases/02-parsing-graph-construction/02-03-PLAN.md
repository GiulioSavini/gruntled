// package hclconv
func Position(file repograph.RepoPath, src []byte, p hcl.Pos) (repograph.Position, error)
// line = 1 + count of '\n' in src[:off]; column = off - (last '\n' before off + 1) + 1, off = clamp(p.Byte, 0, len(src)).
// NEVER uses p.Column (grapheme clusters).
func FirstSyntaxError(file repograph.RepoPath, src []byte, diags hcl.Diagnostics) (diagnostic.Diagnostic, bool, error)
// false when diags has no error-severity entry. Picks the error with the smallest Subject.Start.Byte
// (nil Subject = offset 0), ties broken by Summary then Detail. Message = Summary, plus ": "+Detail when Detail != "".

// package sourceresolve  (stdlib only, no I/O)
type Kind int
const ( KindInvalid Kind = iota; KindLocal; KindRemote )
type Source struct { Kind Kind; Root string; Subdir string } // Root/Subdir set only for KindLocal
func Classify(raw string) Source

// package terragrunt (unexported, same package as the loader of Plans 04-05)
const virtualRoot = "/__gruntled_repo_root__"
type scopeKind int
const ( scopeInclude scopeKind = iota + 1 /* S0: include.path */; scopeUnit /* S1: child body */; scopeIncluded /* S2: inside included file */ )
type includeRef struct { label string; dir string } // every include block of the unit (INCLUDING no_merge), declaration order; dir = repo-relative dir of the included file
type evalScope struct {
    fsys     fs.FS
    unitDir  string       // repo-relative dir of the CHILD unit ("." for the repo root)
    kind     scopeKind
    includes []includeRef // used in S1
    included string       // S2: repo-relative dir of the included file the expression is written in
}
func (s evalScope) functions() map[string]function.Function // exactly the six
func evalPath(expr hcl.Expression, s evalScope) (string, bool)   // raw string value; false on any error/unknown/null/non-string
func resolvePath(unitDir, p string) (string, bool)               // virtual-absolute or relative -> cleaned repo path; false on escape or real absolute path
func literalString(expr hcl.Expression) (string, bool)           // expr.Value(nil): known, non-null cty.String only
func literalBool(expr hcl.Expression) repograph.Tristate         // expr.Value(nil): known non-null cty.Bool -> True/False, anything else -> TristateUnknown
