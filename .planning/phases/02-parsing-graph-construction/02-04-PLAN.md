TARGET internal API (Plan 05's loader consumes exactly this):
```go
// walk.go
type unitEntry struct {
    dir        string // repo-relative unit dir, "." for the repo root
    jsonConfig bool   // terragrunt.hcl.json present (Terragrunt prefers it) => the loader marks the unit json-config-unsupported
}
func discoverUnits(fsys fs.FS) ([]unitEntry, error) // one entry per dir; order = fs.WalkDir order (the loader sorts)

// refs.go
func extractRefs(file repograph.RepoPath, src []byte, body *hclsyntax.Body) ([]repograph.Reference, error) // sorted by byte offset

// depfacts.go
func dependencyOptions(body *hclsyntax.Body) repograph.DependencyOptions

// parse.go
type fileCache struct { fsys fs.FS; files map[string]*parsedFile }
func newFileCache(fsys fs.FS) *fileCache
func (c *fileCache) get(p repograph.RepoPath) *parsedFile // reads + parses at most once per path; never nil
func (c *fileCache) syntaxDiagnostics() []diagnostic.Diagnostic // GRT100 of every file get() touched, sorted by path

type parsedFile struct {
    path    repograph.RepoPath
    src     []byte
    readErr error                   // non-nil => unreadable
    syntax  *diagnostic.Diagnostic  // non-nil => syntax error; NO facts below are populated
    // Facts (only when readErr == nil && syntax == nil), each slice in source order:
    includes   []includeDecl
    terraforms []terraformDecl
    deps       []depDecl
    generates  []generateDecl
    refs       []repograph.Reference
}
type includeDecl struct {
    labels        []string        // 0 labels = bare include; >1 is invalid (loader decides)
    path          hcl.Expression  // nil when the attribute is missing
    mergeStrategy hcl.Expression  // nil when absent (= shallow)
    pos           repograph.Position
}
type terraformDecl struct {
    source hcl.Expression // nil when the block has no source attribute
    pos    repograph.Position
}
type depDecl struct {
    labels       []string        // exactly 1 is valid
    configPath   hcl.Expression  // nil when missing
    hasExpansion bool            // nested `expansion` block (Terragrunt v1.2.0)
    opts         repograph.DependencyOptions
    pos          repograph.Position // block TypeRange.Start
}
type generateDecl struct {
    labels   []string
    contents hcl.Expression // nil when missing
    pos      repograph.Position
}
```
Only TOP-LEVEL blocks of the file body are structural facts (`body.Blocks` is a slice in source order). `dependencies { paths = [...] }` is ordering-only: it is not recorded (DEP-06) but its body is still walked for references like every other block. `body.Attributes` is a map: never range it for anything order-sensitive.
